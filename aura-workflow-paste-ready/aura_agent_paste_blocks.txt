        # `current_project` is conversation state. Leaving the project from the
        # previously open conversation here means a short follow-up such as
        # "continue" can pick up the wrong project's plan and memories. Rebuild
        # it from the most recent user turn in this conversation that actually
        # names a project.
        self.current_project = None
        for item in reversed(messages):
            if item.get("role") != "user":
                continue
            project = self.detect_project(str(item.get("text") or ""))
            if project:
                self.current_project = project
                break
        self.log.record("open_session", "ok", session_id=self.session_id,
                        messages=len(messages))
        return messages

        # state. Do not depend on the model remembering to call
        # `record_plan_steps` in a later round just to make the agreed plan
        # resumable. Existing durable steps win: a later file-plan approval for
        # an established project must not replace work already in progress.

            try:

                    self.db.set_plan_steps(project, listed[:20])
                    self.log.record("plan_steps", "ok", project=project,

            except Exception as exc:
                self._note_bookkeeping_failure("plan_steps", project, exc)
